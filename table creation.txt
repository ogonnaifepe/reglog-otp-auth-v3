ALTER USER getby DEFAULT TABLESPACE RUBIKONDAT01;










-- Connect as GETBY user
-- Create USERS table with all columns including new ones
CREATE TABLE users (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email VARCHAR2(255) UNIQUE NOT NULL,
    password_hash VARCHAR2(255) NOT NULL,
    full_name VARCHAR2(255),
    phone_number VARCHAR2(20),
    profile CHAR(1) DEFAULT 'N',
    customer_no VARCHAR2(20),
    is_verified CHAR(1) DEFAULT 'N',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create OTP table
CREATE TABLE otp_codes (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email VARCHAR2(255) NOT NULL,
    otp_code VARCHAR2(10) NOT NULL,
    purpose VARCHAR2(20) NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    is_used CHAR(1) DEFAULT 'N',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create blacklist tables
CREATE TABLE access_token_blacklist (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    token VARCHAR2(500) NOT NULL,
    email VARCHAR2(255) NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE refresh_token_blacklist (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    token VARCHAR2(500) NOT NULL,
    email VARCHAR2(255) NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create indexes
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_otp_email_purpose ON otp_codes(email, purpose);
CREATE INDEX idx_otp_expires ON otp_codes(expires_at);
CREATE INDEX idx_at_blacklist_token ON access_token_blacklist(token);
CREATE INDEX idx_rt_blacklist_token ON refresh_token_blacklist(token);

-- Grant necessary permissions (if needed)
GRANT SELECT, INSERT, UPDATE, DELETE ON users TO getby;
GRANT SELECT, INSERT, UPDATE, DELETE ON otp_codes TO getby;
GRANT SELECT, INSERT, UPDATE, DELETE ON access_token_blacklist TO getby;
GRANT SELECT, INSERT, UPDATE, DELETE ON refresh_token_blacklist TO getby;
